(defun AM_GetTables( ename / Tables )
(setq Tables (ade_odgettables ename))
(if Tables
(progn
(if (not (listp Tables))
(setq Tables (list Tables))
)
(setq Tables (mapcar 'strcase Tables))
(setq Tables (reverse Tables))
))
Tables
)
